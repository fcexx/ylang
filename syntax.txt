fn -> bin MadeChar(int addr, char symbol){ // мы используем взаместо стандартной типизации (void, int,char ) bin, новый формат для бинарных функций
        mov ax, &addr     // установить текстовый режим 80x25 (очищает экран)
        int 0x10

        mov ah, 0x0E       // функция вывода символа в телетайп-режиме
        mov (al, var(symbol) )       // символ для вывода
        int 0x10            // вызов BIOS

}

        hlt
}
